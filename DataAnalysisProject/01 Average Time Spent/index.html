grouped = df.groupby('platform')
car_owners = grouped['Owns_Car'].mean() * 100
home_owners = grouped['isHomeOwner'].mean() * 100
ownership_distribution = pd.DataFrame({
    'Car Ownership (%)': car_owners,
    'Home Ownership (%)': home_owners
}).round(2)
